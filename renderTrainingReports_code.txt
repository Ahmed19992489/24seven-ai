function renderTrainingReports() {
        const tableBody = document.getElementById('training-reports-table-body');
        const filter = document.getElementById('training-filter-type').value;

        // تصفية البيانات
        const filtered = cachedTrainingReports.filter(r => {
            if (filter === 'all') return true;
            return r.session_type === filter;
        });

        // تحديث العدادات الذكية والـ KPIs
        document.getElementById('training-stat-sessions').textContent = cachedTrainingReports.length;
        
        // حساب متوسط الدرجات (للامتحانات والمحاكاة فقط لاستبعاد الـ 100% التلقائية للكورسات لمصداقية النسبة)
        const activeScores = cachedTrainingReports.filter(r => r.session_type !== 'course');
        if (activeScores.length > 0) {
            const avg = Math.round(activeScores.reduce((s, r) => s + (r.score || 0), 0) / activeScores.length);
            document.getElementById('training-stat-avg').textContent = `${avg}%`;
        } else {
            document.getElementById('training-stat-avg').textContent = `--`;
        }

        // حساب الموظف الأعلى تقييماً
        const bestEmpMap = {};
        cachedTrainingReports.forEach(r => {
            if (!bestEmpMap[r.employee_name] || bestEmpMap[r.employee_name] < r.score) {
                bestEmpMap[r.employee_name] = r.score;
            }
        });
        let bestScore = -1;
        let bestName = "--";
        for (let name in bestEmpMap) {
            if (bestEmpMap[name] > bestScore) {
                bestScore = bestEmpMap[name];
                bestName = `${name} (${bestScore}%)`;
            }
        }
        document.getElementById('training-stat-best').textContent = bestName;

        if (filtered.length === 0) {
            tableBody.innerHTML = '<tr><td colspan="5" class="text-center py-6 text-slate-400">لا توجد تقارير مطابقة للفحص الحالي.</td></tr>';
            return;
        }

        tableBody.innerHTML = filtered.map(r => {
            const dateStr = new Date(r.created_at).toLocaleString('ar-EG');
            
            // بادج نوع الجلسة
            const typeBadge = {
                roleplay: '<span class="px-2 py-1 rounded-full bg-amber-50 text-amber-700 font-bold border border-amber-200">🎭 محاكاة عميل</span>',
                course: '<span class="px-2 py-1 rounded-full bg-indigo-50 text-indigo-700 font-bold border border-indigo-200">📖 كورس تعليمي</span>',
                exam: '<span class="px-2 py-1 rounded-full bg-teal-50 text-teal-700 font-bold border border-teal-200">📝 اختبار نهائي</span>'
            }[r.session_type] || r.session_type;

            // بادج النتيجة
            let scoreClass = "bg-rose-50 text-rose-700 border-rose-200";
            if (r.score >= 90) scoreClass = "bg-emerald-50 text-emerald-700 border-emerald-200";
            else if (r.score >= 70) scoreClass = "bg-amber-50 text-amber-700 border-amber-200";

            const scoreBadge = `<span class="px-2 py-1 rounded-full font-bold border ${scoreClass}">${r.score} / 100</span>`;

            return `
                <tr class="hover:bg-slate-50 transition border-b border-slate-100">
                    <td class="p-3 font-bold text-slate-800">${r.employee_name}</td>
                    <td class="p-3">${typeBadge}</td>
                    <td class="p-3">${scoreBadge}</td>
                    <td class="p-3 text-slate-400 font-mono">${dateStr}</td>
                    <td class="p-3">
                        <button onclick="viewTrainingDetail('${r.id}')" class="bg-indigo-600 hover:bg-indigo-700 text-white font-bold py-1 px-3 rounded-lg text-[10px] transition flex items-center gap-1">
                            <i class="fas fa-eye"></i> عرض المحادثة والتقرير
                        </button>
                    </td>
                </tr>
            `;
        }).join('');
    }
