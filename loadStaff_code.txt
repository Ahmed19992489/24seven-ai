async function loadStaff() {
            const list = document.getElementById('staff-table-body');
            list.innerHTML = '<tr><td colspan="5" class="text-center p-10 text-slate-400"><i class="fas fa-spinner fa-spin text-2xl mb-2"></i><br>جاري التحميل...</td></tr>';
            
            try {
                // جلب الموظفين والمديرين من جدول profiles
                const { data: staff, error, count } = await sbClient
                    .from('profiles')
                    .select('*', { count: 'exact' })
                    .in('role', ['admin', 'moderator'])
                    .not('email', 'is', null)
                    .order('created_at', { ascending: false });

                if (error) throw error;

                document.getElementById('staff-total-count').innerText = count || 0;

                if (!staff || staff.length === 0) {
                    list.innerHTML = '<tr><td colspan="5" class="text-center p-10 text-slate-400">لا يوجد موظفين مسجلين حالياً</td></tr>';
                    return;
                }

                list.innerHTML = staff.map(s => {
                    const statusInfo = getOnlineStatus(s.last_seen);
                    return `
                    <tr class="hover:bg-slate-50 transition">
                        <td class="p-4 font-bold text-slate-700">${escapeHtml(s.full_name || 'بدون اسم')}</td>
                        <td class="p-4 text-slate-500">${escapeHtml(s.email || '-')}</td>
                        <td class="p-4">
                            <span class="px-3 py-1 rounded-full text-[10px] font-bold ${s.role === 'admin' ? 'bg-red-100 text-red-600' : 'bg-blue-100 text-blue-600'} uppercase">
                                ${s.role === 'admin' ? 'مدير نظام' : 'موظف (Mod)'}
                            </span>
                        </td>
                        <td class="p-4">
                            <div class="flex items-center gap-2">
                                <span class="w-2 h-2 rounded-full ${statusInfo.online ? 'bg-green-500 animate-pulse' : 'bg-slate-300'}"></span>
                                <span class="text-[10px] font-bold ${statusInfo.online ? 'text-green-600' : 'text-slate-400'}">${statusInfo.text}</span>
                            </div>
                        </td>
                        <td class="p-4 text-[11px] text-slate-400">${s.created_at ? new Date(s.created_at).toLocaleDateString('ar-EG') : '-'}</td>
                        <td class="p-4 flex gap-2">
                             <button onclick="deleteStaff('${s.id}', '${escapeHtml(s.full_name)}')" class="text-slate-400 hover:text-red-500 transition"><i class="fas fa-trash-alt"></i></button>
                        </td>
                    </tr>
                `}).join('');

            } catch (err) {
                console.error('Staff Load Error:', err);
                list.innerHTML = `<tr><td colspan="5" class="text-center p-10 text-red-500">❌ فشل تحميل القائمة: ${err.message}</td></tr>`;
            }
        }
